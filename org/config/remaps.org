            (use-package general 
              :ensure t 
          	  :demand t
              :config
              (general-evil-setup)
              (defconst leader-def "SPC")
              (general-create-definer leader
            			      :states 'normal
            			      :keymaps 'override
            			      :prefix leader-def)
              (leader
          	 "f" '(:ignore t)
      	     "f f" 'consult-find
      	     "f g" 'consult-ripgrep
      	     "f b" 'consult-buffer
      	     "f h" 'consult-man
          	 "f c" '(lambda () (interactive) (dired user-emacs-directory))
           )
          (leader
          	 "r" '(:ignore t)
          	 "r r" '(lambda () (interactive) (load-file user-init-file))
      	     "r u" 'elpaca-update-all
           )
      (leader
              	 "t" '(:ignore t)
              "t l" 'display-line-numbers-mode
      	 "t t" 'visual-line-mode
      	 "t o" 'org-mode
      	 "t m" 'toggle-markdown-view-mode
      	 "t w" 'writeroom-mode
          )
      (leader
        "o" '(:ignore t)
        "o f" 'org-roam-node-find
        "o i" 'org-roam-node-insert
        "o g" 'org-roam-ui-open
    ))
#+end_src
